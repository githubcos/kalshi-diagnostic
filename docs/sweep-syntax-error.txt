		baseline.ObjectiveScore,
	)

	sort.Slice(results, func(i, j int) bool {
		if results[i].ObjectiveScore == results[j].ObjectiveScore {
			if results[i].Net == results[j].Net {
				return results[i].MaxDrawdown < results[j].MaxDrawdown
			}
			return results[i].Net > results[j].Net
		}
		return results[i].ObjectiveScore > results[j].ObjectiveScore
	})

	topN := 20
	if len(results) < topN {
		topN = len(results)
	}
	for i := 0; i < topN; i++ {
		r := results[i]
		t.Logf("v3 rank=%d score=%.3f net=%.2f dd=%.2f trades=%d win_rate=%.2f%% losses=%d stop=%d safe=%d timeout=%d cfg={minVel=%s minLock=%s cvd=%s cbSpread=%s cbTI=%s minOI=%s maxContraOI=%s}",
			i+1,
			r.ObjectiveScore,
			r.Net,
			r.MaxDrawdown,
			r.Trades,
			r.WinRate,
			r.Losses,
			r.StopLossCount,
			r.UnhedgedSafeExitCount,
			r.UnhedgedTimeoutCount,
			r.Config["PAIR_ARB_MIN_GAP_VELOCITY_USD"],
			r.Config["PAIR_ARB_MIN_LOCKED_PROFIT_CENTS"],
			r.Config["PAIR_ARB_MIN_CVD_BTC"],
			r.Config["PAIR_ARB_MIN_COINBASE_SPREAD_USD"],
			r.Config["PAIR_ARB_MIN_COINBASE_TAKER_IMBALANCE"],
			r.Config["PAIR_ARB_MIN_OI_DELTA_USD"],
			r.Config["PAIR_ARB_MAX_CONTRA_OI_DELTA_USD"],
		)
	}
}
